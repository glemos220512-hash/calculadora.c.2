```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int main() {
    int opcao;
    double num1, num2, resultado;
    char continuar;
    char entrada[100];
    char extra;

    do {
        printf("\n=============================\n");
        printf("       Calculadora Simples\n");
        printf("=============================\n");
        printf("Selecione uma operacao:\n");
        printf("1. Adicao\n");
        printf("2. Subtracao\n");
        printf("3. Multiplicacao\n");
        printf("4. Divisao\n");
        printf("5. Sair\n");

        /* Validacao da opcao */
        while (1) {
            printf("Opcao: ");

            if (fgets(entrada, sizeof(entrada), stdin) == NULL) {
                return 0;
            }

            /* Verifica se existe exatamente um inteiro na entrada */
            if (sscanf(entrada, " %d %c", &opcao, &extra) == 1 &&
                opcao >= 1 && opcao <= 5) {
                break;
            }

            printf("Opcao invalida! Digite um numero inteiro entre 1 e 5.\n");
        }

        /* Opcao sair */
        if (opcao == 5) {
            printf("Obrigado por usar a calculadora!\n");
            break;
        }

        /* Entrada dos numeros */
        printf("Digite o primeiro numero: ");
        scanf("%lf", &num1);

        printf("Digite o segundo numero: ");
        scanf("%lf", &num2);

        /* Limpa o restante da entrada */
        while (getchar() != '\n');

        /* Operacoes */
        switch (opcao) {
            case 1:
                resultado = num1 + num2;
                printf("Resultado: %.2f + %.2f = %.2f\n",
                       num1, num2, resultado);
                break;

            case 2:
                resultado = num1 - num2;
                printf("Resultado: %.2f - %.2f = %.2f\n",
                       num1, num2, resultado);
                break;

            case 3:
                resultado = num1 * num2;
                printf("Resultado: %.2f * %.2f = %.2f\n",
                       num1, num2, resultado);
                break;

            case 4:
                if (num2 == 0) {
                    printf("Erro: Divisao por zero!\n");
                } else {
                    resultado = num1 / num2;
                    printf("Resultado: %.2f / %.2f = %.2f\n",
                           num1, num2, resultado);
                }
                break;
        }

        /* Pergunta se deseja continuar */
        while (1) {
            printf("Deseja realizar outra operacao? (s/n): ");

            if (scanf(" %c", &continuar) != 1) {
                return 0;
            }

            while (getchar() != '\n');

            if (continuar == 's' || continuar == 'S' ||
                continuar == 'n' || continuar == 'N') {
                break;
            }

            printf("Resposta invalida! Digite 's' ou 'n'.\n");
        }

        if (continuar == 'n' || continuar == 'N') {
            printf("Obrigado por usar a calculadora!\n");
            break;
        }

    } while (continuar == 's' || continuar == 'S');

    return 0;
}
